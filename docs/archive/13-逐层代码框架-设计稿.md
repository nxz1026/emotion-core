```python
    PROMOTION_MIN_DENOM: int = 3               # 分母不足 → None，不补零（F6）
    # ── 运维 ──────────────────────────────────

    # ── 开关 ──────────────────────────────────
    SECONDARY_EXPORT_ENABLED: bool = False     # 次级推荐影子模式
    TRADE_EXPORT_ENABLED: bool = False         # 交易桥默认不导出
    LLM_PROFILE: str | None = None             # None=关；'agnes'=启用

CONFIG = Config()

def config_hash() -> str:
    """每次运行写入 pipeline_state / signal / eval_result（S9）。"""
    payload = json.dumps(asdict(CONFIG), default=str, sort_keys=True)
    return hashlib.sha256(payload.encode()).hexdigest()[:12]
```

**price.py**（全项目唯一涨停价实现，守护测试锁定）：

```python
"""涨停价唯一实现。单位：分（int）。板块感知：主板10% / 创业30 科创68→20% / 北交所bj→30%。

口径来源：lkl utils/price.py:9（Decimal 参考）+ derive.py:48-52（SQL 整数式，生产用）。
两者语义等价；本实现用整数分运算，与 SQL 整数式逐分一致（S5）。
"""

def board_pct_milli(code: str) -> int:
    """返回涨跌幅的千分比（主板 100 = 10.0%）。"""
    if code.startswith("bj"):                 return 300   # 北交所 30%
    if code.startswith(("30", "68")):         return 200   # 创业/科创 20%
    return 100                                            # 主板 10%

def limit_up_price_cents(pre_close_cents: int, code: str) -> int:
    """涨停价（分）。四舍五入与 SQL (x*pct + 50)/100 一致。"""
    pct = board_pct_milli(code)
    return (pre_close_cents * (1000 + pct) + 5000) // 10000

def is_limit_up(close_cents: int, pre_close_cents: int, code: str) -> bool:
    return close_cents >= limit_up_price_cents(pre_close_cents, code)
```

**纪律**（照 `docs/07` §9.4）：
- 只收 POD（`i64` 分、`&[BarIn]`），不做任何 I/O；
- 连板实现要点（**最大风险点**，`docs/07` §2.2）：按 `(code, date)` 排序后**只遍历实际存在的行**，遇到非涨停行重置计数——停牌日无行，自然实现「停牌断档不打断」。**不按日历补齐**；
- 状态机：`_RULES` 优先级数组 `退潮 > 高潮 > 发酵 > 冰点`，首个命中定相；全未命中 → 继承昨日（`phase_inherited=true`），首日 seed 冰点；
- entry 的 c3 淘汰性：**集合相等判定**（`survivor_set == {candidate}`），精确保留；
- 单元测试在 Rust 侧用 `#[cfg(test)]`，每个模块至少一个 lkl 取证样例。

> S1 的「Python 先行」若被采纳，本层每个函数对应一个已通过 lkl 对账的 Python 参考实现，Rust 测试直接以 Python 输出为 oracle。

```python
from emotion_core.algorithms.state import classify_series, EmotionState, DayMetrics

def run_state_machine(metrics: list[DayMetrics]) -> list[EmotionState]:
    """4 相状态机（逐日推进）。语义逐字照 lkl emotion.py:classify。
```

```python
def run_day(target: date) -> EmotionState:
    bars = data.db.load_bars(target)                      # domain.Bar（分）
    bars = [b for b in bars if not b.code.startswith("bj")]      # C7：判定口径冻结不含北交所
    bars = [b for b in bars if not data.db.is_st(b.code, target)]  # D1：剔 ST

    data.db.upsert_market_stat(target, result, config_hash=utils.config.config_hash())
    return result
```
